    <script>
        // --- 1. Global Parameters & Constants ---
        let L = 1500;   // Carrying capacity (g)
        let k = 0.18;   // Growth coefficient
        let x0 = 20;    // Inflection point (week)
        let activeX = 20; // Currently selected week index/value
        let animationInterval = null;

        // Color palettes
        const EMERALD_COLOR = 'rgba(5, 150, 105, 0.9)';
        const TEAL_LIGHT_COLOR = 'rgba(45, 212, 191, 0.7)';
        const TANGENT_COLOR = 'rgba(239, 68, 68, 1)'; // Rose red for clear geometry tangent

        // --- 2. Bioscience Math Solver Functions ---
        // f(x) Logistic curve function
        function f(x, pL = L, pK = k, pX0 = x0) {
            return pL / (1 + Math.exp(-pK * (x - pX0)));
        }

        // f'(x) Analytical First Derivative
        function fPrime(x, pL = L, pK = k, pX0 = x0) {
            const expTerm = Math.exp(-pK * (x - pX0));
            const denominator = Math.pow(1 + expTerm, 2);
            return (pL * pK * expTerm) / denominator;
        }

        // Get tangent points based on selected coordinate
        // Generates an array of points for ChartJS representing a linear tangent line localized around point 'xCoord'
        function getTangentDatasetPoints(xCoord, pL = L, pK = k, pX0 = x0) {
            const yCoord = f(xCoord, pL, pK, pX0);
            const slope = fPrime(xCoord, pL, pK, pX0);
            
            const linePoints = [];
            // We want the tangent line to span a local window (e.g., activeX - 6 to activeX + 6) 
            const windowSize = 6;
            const minX = Math.max(0, xCoord - windowSize);
            const maxX = Math.min(45, xCoord + windowSize);

            for (let x = 0; x <= 45; x++) {
                if (x >= minX && x <= maxX) {
                    // Line equation: y = m(x - x_1) + y_1
                    const y = slope * (x - xCoord) + yCoord;
                    linePoints.push({ x: x, y: y });
                } else {
                    linePoints.push({ x: x, y: null }); // Keep rest of points null to restrict rendering span
                }
            }
            return linePoints;
        }

        // --- 3. Interactive Data Generator ---
        function generateChartData() {
            const labels = [];
            const biomassData = [];
            const growthRateData = [];

            for (let week = 0; week <= 45; week++) {
                labels.push(week);
                biomassData.push(f(week).toFixed(1));
                growthRateData.push(fPrime(week).toFixed(2));
            }

            return { labels, biomassData, growthRateData };
        }

        // --- 4. ChartJS Implementation ---
        let growthChart;
        const ctx = document.getElementById('growthChart').getContext('2d');

        function initChart() {
            const initialData = generateChartData();
            const tangentPoints = getTangentDatasetPoints(activeX);

            growthChart = new Chart(ctx, {

                            borderWidth: 2,

                            order: 3

        // --- 5. Interactive Update Master Sync ---
        function updateActiveTime(week) {
            activeX = parseInt(week);
            if (activeX < 0) activeX = 0;
            if (activeX > 45) activeX = 45;

            // Update Tangent Dataset on Chart
            const newTangent = getTangentDatasetPoints(activeX);
            growthChart.data.datasets[2].data = newTangent;
            growthChart.update('none'); // Quick update without disturbing user's visual continuity

            // Calculate calculations for Dashboard
            const currentBiomass = f(activeX);
            const currentRate = fPrime(activeX);

            // Populate calculations in UI
            document.getElementById('diagWeek').innerText = `Week ${activeX}`;
            document.getElementById('diagBiomass').innerText = currentBiomass.toFixed(1);
            document.getElementById('diagRate').innerText = currentRate.toFixed(2);

            // Update Dynamic SVGs and agronomy text
            updateSVGVisualization(activeX, currentBiomass);
            updateAdvisoryTexts(activeX, currentBiomass, currentRate);
        }

        // Dynamic change of Pineapple crop model size and fruiting triggers
        function updateSVGVisualization(week, biomass) {
            const scaleFactor = 0.3 + (biomass / L) * 0.7; // Maps growth to vector scale dynamically
            const plantGroup = document.getElementById('plantTransform');
            const fruitGroup = document.getElementById('fruitGroup');
            
            // Scaled matrix
            plantGroup.setAttribute('transform', `scale(${scaleFactor}) translate(50, ${100 - (scaleFactor * 40)})`);

            // Phase and color triggers based on plant vegetative age
            const leafOuterL = document.getElementById('leafOuterL');
            const leafOuterR = document.getElementById('leafOuterR');
            const leafMidL = document.getElementById('leafMidL');
            const leafMidR = document.getElementById('leafMidR');
            const leafInner = document.getElementById('leafInner');
            const phaseText = document.getElementById('visualPhaseText');

            if (week <= 12) {
                // Young establishment
                leafOuterL.setAttribute('fill', '#a7f3d0');
                leafOuterR.setAttribute('fill', '#a7f3d0');
                leafMidL.setAttribute('fill', '#34d399');
                leafMidR.setAttribute('fill', '#34d399');
                leafInner.setAttribute('fill', '#059669');
                fruitGroup.setAttribute('transform', 'translate(50, 60) scale(0)');
                phaseText.innerText = "Establishment Phase (Slow Growth)";
                phaseText.className = "text-xs font-bold text-teal-600";
            } else if (week > 12 && week <= 28) {
                // High growth vegetative expansion
                leafOuterL.setAttribute('fill', '#34d399');
                leafOuterR.setAttribute('fill', '#34d399');
                leafMidL.setAttribute('fill', '#059669');
                leafMidR.setAttribute('fill', '#059669');
                leafInner.setAttribute('fill', '#047857');
                fruitGroup.setAttribute('transform', 'translate(50, 60) scale(0)');
                phaseText.innerText = "Logarithmic Surge Phase (Fast Growth)";
                phaseText.className = "text-xs font-bold text-amber-600 animate-pulse";
            } else if (week > 28) {
                // Hormonal/Flower Induction (Ethrel induction window begins)
                leafOuterL.setAttribute('fill', '#047857');
                leafOuterR.setAttribute('fill', '#047857');
                leafMidL.setAttribute('fill', '#065f46');
                leafMidR.setAttribute('fill', '#065f46');
                leafInner.setAttribute('fill', '#022c22');
                
                // Show fruit scaling up with reproductive timeline
                const fruitScale = Math.min(1.2, (week - 28) / 12);
                fruitGroup.setAttribute('transform', `translate(0, -35) scale(${fruitScale})`);
                
                if (week < 36) {
                    phaseText.innerText = "Inflorescence & Flower Development";
                    phaseText.className = "text-xs font-bold text-rose-600";
                } else {
                    phaseText.innerText = "Fruit Maturation & Ripening";
                    phaseText.className = "text-xs font-bold text-yellow-600";
                }
            }
        }

        // Animation Timeline Loop
        const btnPlay = document.getElementById('btnPlay');
        const btnStop = document.getElementById('btnStop');

        function startTimelineAnimation() {
            btnPlay.classList.add('hidden');
            btnStop.classList.remove('hidden');
            
            let animWeek = 0;
            updateActiveTime(animWeek);

            animationInterval = setInterval(() => {
                animWeek++;
                if (animWeek > 45) {
                    stopTimelineAnimation();
                } else {
                    updateActiveTime(animWeek);
                }
            }, 180); // Smooth stepping across time
        }

        function stopTimelineAnimation() {
            clearInterval(animationInterval);
            btnPlay.classList.remove('hidden');
            btnStop.classList.add('hidden');
        }

        btnPlay.addEventListener('click', startTimelineAnimation);
        btnStop.addEventListener('click', stopTimelineAnimation);
